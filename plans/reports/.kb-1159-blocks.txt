# ═══ Sub-step 4: Pre-populate imfile state files với offset=EOF ═══
# BẮT BUỘC cho Loại a — rsyslog v8.24 không có freshStartTail. Nếu bỏ qua,
# imfile burst đọc 4-8 GB → CPU 78%+ và queue overflow.
for src in mailbox.log audit.log zimbra.log; do
  SL=/var/log/onelog-ship/$src
  INODE=$(stat -Lc %i "$SL")
  SIZE=$(stat -Lc %s "$SL")
  # Xóa state hash-suffix cũ liên quan symlink path
  grep -l "\"filename\": \"$SL\"" /var/lib/rsyslog/imfile-state:* 2>/dev/null | xargs -r rm -f
  printf '{ "filename": "%s", "prev_was_nl": 0, "curr_offs": %d, "strt_offs": %d }' \
    "$SL" "$SIZE" "$SIZE" > "/var/lib/rsyslog/imfile-state:$INODE"
done
